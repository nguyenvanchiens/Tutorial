Protected Route

function RequireAuth({ children }) {
  const user = useAuth();
  const location = useLocation();

  if (!user) {
    return <Navigate
      to="/login"
      state={{ from: location }}
      replace />;
  }
  return children;
}

Zustand — 2026 khuyên mặc định

import { create } from 'zustand';

const useCart = create((set) => ({
  items: [],
  add: (item) => set(state => ({ items: [...state.items, item] })),
  remove: (id) => set(state => ({ items: state.items.filter(x => x.id !== id) })),
  clear: () => set({ items: [] }),
}));

// Dùng trong component — chỉ re-render khi selector đổi
function Cart() {
  const items = useCart(s => s.items);
  const remove = useCart(s => s.remove);
  return ...;
}

🔌 Data Fetching — React Query

Dừng viết useEffect + fetch + loading + error. Dùng TanStack React Query — caching, retry, revalidation miễn phí.

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

// Fetch
function Products() {
  const { data, isLoading, error } = useQuery({
    queryKey: ['products'],
    queryFn: () => fetch('/api/products').then(r => r.json()),
    staleTime: 60_000,  // cache 60s trước khi refetch
  });

  if (isLoading) return <Spinner />;
  if (error) return <Error />;
  return <List items={data} />;
}

// Mutate + invalidate cache
function AddForm() {
  const qc = useQueryClient();
  const { mutate } = useMutation({
    mutationFn: (newItem) => fetch('/api/products', {
      method: 'POST',
      body: JSON.stringify(newItem),
    }),
    onSuccess: () => qc.invalidateQueries({ queryKey: ['products'] }),
  });
  ...
}
Lợi ích miễn phí:
  • Tự cache theo queryKey, share giữa component
  • Tự refetch khi focus window, reconnect mạng
  • Dedupe request trùng, retry khi fail
  • Optimistic update
  • Pagination, infinite query sẵn API

Lib tương đương: SWR (Vercel, nhẹ hơn), RTK Query (nếu đã dùng Redux).

📝 Forms — React Hook Form + Zod

Form phức tạp (validate, arrays, multi-step) → RHF nhanh hơn controlled 5x, validate bằng Zod schema.

import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';

const schema = z.object({
  email: z.string().email('Email sai định dạng'),
  age:   z.number().min(18, 'Phải đủ 18'),
});

function RegisterForm() {
  const {
    register, handleSubmit, formState: { errors, isSubmitting }
  } = useForm({ resolver: zodResolver(schema) });

  const onSubmit = async (data) => {
    await api.register(data);
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register('email')} />
      {errors.email && <span>{errors.email.message}</span>}

      <input type="number" {...register('age', { valueAsNumber: true })} />
      {errors.age && <span>{errors.age.message}</span>}

      <button disabled={isSubmitting}>Đăng ký</button>
    </form>
  );
}
Sức mạnh Zod: 1 schema — dùng cho client validate, API contract, infer TypeScript type (type User = z.infer<typeof schema>).

🧪 Testing

Test pyramid cho React: nhiều unit, vừa phải integration, ít E2E. Focus vào hành vi, không implementation.

Vitest + React Testing Library

import { render, screen, userEvent } from '@testing-library/react';
import Counter from './Counter';

test('tăng đếm khi click', async () => {
  render(<Counter />);
  const btn = screen.getByRole('button', { name: /\+/i });
  await userEvent.click(btn);
  expect(screen.getByText('1')).toBeInTheDocument();
});
Mantra của RTL: "Test app như user dùng". Query theo role, label, text, không theo className/id.
